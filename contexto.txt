import { useState } from 'react';
import {
  View,
  Text,
  ScrollView,
  KeyboardAvoidingView,
  Pressable,
  StyleSheet,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import TextField from '../components/TextField';
import Button from '../components/Button';
import useAuth from '../hooks/useAuth';
import { validateEmail, validatePassword } from '../utils/validations';
import { colors, spacing, typography } from '../theme';
import { t } from '../i18n';

export default function LoginScreen({ navigation }) {
  const { signIn } = useAuth();

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [errors, setErrors] = useState({ email: '', password: '' });
  const [isLoading, setIsLoading] = useState(false);
  const handleEmailChange = (text) => {
    setEmail(text);
    if (errors.email) setErrors((prev) => ({ ...prev, email: '' }));
  };

  const handlePasswordChange = (text) => {
    setPassword(text);
    if (errors.password) setErrors((prev) => ({ ...prev, password: '' }));
  };

  const handleSubmit = async () => {
    if (isLoading) return;

    const newErrors = {
      email: validateEmail(email),
      password: validatePassword(password),
    };
    setErrors(newErrors);
    if (newErrors.email || newErrors.password) return;

    setIsLoading(true);
    await new Promise((resolve) => setTimeout(resolve, 800));
    signIn();
  };

  return (
    <SafeAreaView style={styles.screen}>
      <KeyboardAvoidingView style={styles.flex} behavior="padding">
        <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
          <Text style={styles.title}>{t.login.title}</Text>

          <TextField
            label={t.login.email}
            placeholder={t.login.emailPlaceholder}
            value={email}
            onChangeText={handleEmailChange}
            onBlur={() => setErrors((prev) => ({ ...prev, email: validateEmail(email) }))}
            error={errors.email}
            keyboardType="email-address"
            autoCapitalize="none"
            autoComplete="email"
          />

          <TextField
            label={t.login.password}
            placeholder={t.login.passwordPlaceholder}
            value={password}
            onChangeText={handlePasswordChange}
            onBlur={() => setErrors((prev) => ({ ...prev, password: validatePassword(password) }))}
            error={errors.password}
            isPassword
            autoCapitalize="none"
          />

          <Button
            title={isLoading ? t.login.submitting : t.login.submit}
            onPress={handleSubmit}
            disabled={isLoading}
          />

          <View style={styles.linkRow}>
            <Text style={styles.linkText}>{t.login.noAccount} </Text>
            <Pressable onPress={() => navigation.navigate('Register')} hitSlop={8}>
              <Text style={styles.link}>{t.login.signUpLink}</Text>
            </Pressable>
          </View>
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

                  display={Platform.OS === 'ios' ? 'spinner' : 'default'}

                  </Pressable>
                )}
              </View>
            )}

            {!!personalErrors.birthDate && (
              <Text style={styles.error}>
                {t.errors[personalErrors.birthDate] ??
                  personalErrors.birthDate}
              </Text>
            )}
          </View>

          <Dropdown
            label="Género *"
            value={personalData.gender}
            options={genderOptions}
            onChange={(value) => updateField('gender', value)}
            onTouch={() => touchPersonal('gender')}
            error={personalErrors.gender}
          />

          <TextField
            label="Correo electrónico *"
            placeholder="conductor@correo.com"
            value={personalData.email}
            onChangeText={(text) => updateField('email', text)}
            onBlur={() => touchPersonal('email')}
            error={personalErrors.email}
            keyboardType="email-address"
            autoCapitalize="none"
            autoComplete="email"
          />

          <Dropdown
            label="Idioma *"
            value={personalData.language}
            options={languageOptions}
            onChange={(value) => updateField('language', value)}
            onTouch={() => touchPersonal('language')}
            error={personalErrors.language}
          />

          <Button title="Validar datos personales" onPress={handleValidate} />

          {!!message && (
            <Text style={styles.success} accessibilityLiveRegion="polite">
              {message}
            </Text>
          )}
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

import { createNativeStackNavigator } from '@react-navigation/native-stack';

import WelcomeScreen from '../screens/WelcomeScreen';
import LoginScreen from '../screens/LoginScreen';
import HomeScreen from '../screens/HomeScreen';
import PersonalDataScreen from '../screens/registro/PersonalDataScreen';

import useAuth from '../hooks/useAuth';
import { colors } from '../theme';
import { t } from '../i18n';

const Stack = createNativeStackNavigator();

export default function AppNavigator() {
  const { isSignedIn } = useAuth();

  return (
    <Stack.Navigator
      screenOptions={{
        headerStyle: {
          backgroundColor: colors.background,
        },
        headerTintColor: colors.text,
        headerTitleStyle: {
          fontWeight: '700',
        },
        contentStyle: {
          backgroundColor: colors.background,
        },
      }}
    >
      {isSignedIn ? (
        <Stack.Screen
          name="Home"
          component={HomeScreen}
          options={{
            title: t.home.title,
            headerBackVisible: false,
            gestureEnabled: false,
          }}
        />
      ) : (
        <>
          <Stack.Screen
            name="Welcome"
            component={WelcomeScreen}
            options={{ headerShown: false }}
          />

          <Stack.Screen
            name="Login"
            component={LoginScreen}
            options={{ title: 'Iniciar sesión' }}
          />

          <Stack.Screen
            name="Register"
            component={PersonalDataScreen}
            options={{ title: 'Registro del conductor' }}
          />
        </>
      )}
    </Stack.Navigator>
  );
}
